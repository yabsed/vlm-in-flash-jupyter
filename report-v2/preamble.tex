% Chapter: 제 1 장 제목, centered, 16pt
\titleformat{\chapter}[block]
  {\centering\normalfont\fontsize{16pt}{16pt}\bfseries} % Centered, bold font
  {제 \thechapter{} 장}                                  % Chapter number format
  {10pt}                                                 % Space between number and title
  {}

\titlespacing*{\chapter}
  {0pt}    % Indentation from the left margin
  {-30pt}  % Space before the chapter title (negative value to reduce)
  {20pt}   % Space after the chapter title

% Section: 제 1 절 제목, 14pt
\titleformat{\section}
  {\normalfont\fontsize{14pt}{14pt}\bfseries}
  {제 \arabic{section} 절}
  {10pt}
  {}

% Subsection: 1.1.1 제목, 12pt
\titleformat{\subsection}
  {\normalfont\fontsize{12pt}{12pt}\bfseries}
  {\thesubsection}
  {10pt}
  {}

% Table of contents: the same numbering as the headings
\newcommand{\LastNumber}[1]{\expandafter\TakeLastNumber#1\relax}

\titlecontents{subsection}[3.5em]

  {\thecontentslabel\quad}
  {}
  {\titlerule*[0.6em]{.}\contentspage}
